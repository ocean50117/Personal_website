看 Python 原型（終端機版）
import datetime

# 1. 讓使用者輸入生日,格式是 yyyy-mm-dd
birth_input = input("Enter your date of birth (yyyy-mm-dd): ")

# 2. 用 .split() 把字串拆成三段,再用 int() 把每一段轉成數字
birth_parts = birth_input.split("-")
birth_year = int(birth_parts[0])
birth_month = int(birth_parts[1])
birth_day = int(birth_parts[2])

# 3. 建立生日的 date 物件
birthday = datetime.date(birth_year, birth_month, birth_day)

# 4. 計算活了幾天、幾週
today = datetime.date.today()
days_lived = (today - birthday).days
weeks_lived = days_lived // 7

# 5. 設定總共要畫幾行、每行幾格
total_years = 90
weeks_per_year = 52

print(f"\nYou have lived {weeks_lived} weeks so far.\n")

# 6. 用巢狀迴圈畫出 90 行 x 52 格的方塊圖
for row in range(total_years):
    line = ""
    for col in range(weeks_per_year):
        week_number = row * weeks_per_year + col
        if week_number < weeks_lived:
            line += "■ "
        else:
            line += "□ "
    print(line)

Day 7・生日計算器與生命倒數

用 datetime 計算下一個生日、已經活了多久，並自己寫一個模組放隨機祝福語，開始用 GitHub 管理程式。看貼文 ↗

看程式碼
import datetime, bday_messages          # 內建 datetime + 自己寫的模組

today = datetime.date.today()
next_birthday = datetime.date(YYYY, M, D)      # 下一個生日
days_away = (next_birthday - today).days

if today == next_birthday:
    print(bday_messages.random_message)        # 生日當天隨機一句祝福
else:
    print(f'My next birthday is {days_away} days away!')

day_lived = (today - datetime.date(YYYY, M, D)).days
years = day_lived // 365
months = (day_lived % 365) // 30
print(f'I have lived for {years} years and {months} months!')

Day 17・自製 Discord 迷因機器人

輸入 $meme，機器人就從 API 抓一張隨機迷因回傳。練習事件驅動、類別繼承與 JSON，並延伸思考它跟 MCP 的差異。看 LinkedIn 影片 ↗

看程式碼
import discord            # discord.py 客戶端庫 (非同步)
import requests           # 同步 HTTP 請求庫
import json               # JSON 解析

def get_meme():
  # 向 meme-api 要一張隨機 meme，回傳 JSON 中的 'url' 欄位
  response = requests.get('https://meme-api.com/gimme')
  json_data = json.loads(response.text)
  return json_data['url']

class MyClient(discord.Client):
  async def on_ready(self):
    # 當 bot 成功連線並準備好時呼叫
    print('Logged on as {0}!'.format(self.user))

  async def on_message(self, message):
    # 忽略自己發的訊息，避免回圈
    if message.author == self.user:
      return
    # 如果訊息以 $meme 開頭，就取得 meme 並回傳到該頻道
    if message.content.startswith('$meme'):
      await message.channel.send(get_meme())

# 啟用 intents。discord.py 需要顯式開啟 message_content 才能讀取訊息內容
intents = discord.Intents.default()
intents.message_content = True

# 建立並啟動客戶端
client = MyClient(intents=intents)
client.run(TOKEN)  # token 放環境變數，不寫進程式碼
